add_library(launcher-core STATIC core.cpp platform.cpp bundle.cpp runtime.cpp self_test.cpp "${LAUNCHER_DEPS}/miniz-3.1.2/miniz_tinfl.c")
target_include_directories(launcher-core PUBLIC "${LAUNCHER_DEPS}" "${LAUNCHER_DEPS}/miniz-3.1.2" "${OPENXR_INCLUDE}")
target_compile_definitions(launcher-core PUBLIC DTVR_VERSION="${PROJECT_VERSION}" NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS MINIZ_NO_ARCHIVE_APIS MINIZ_NO_DEFLATE_APIS MINIZ_NO_STDIO MINIZ_NO_TIME)
if(WIN32)
  target_compile_definitions(launcher-core PUBLIC UNICODE _UNICODE)
  add_executable(dirttown-vr-launcher WIN32 main.cpp gui_win.cpp)
  target_link_libraries(launcher-core PUBLIC advapi32 shell32)
  target_link_libraries(dirttown-vr-launcher PRIVATE launcher-core ole32 uuid comctl32 gdi32)
  if(MINGW)
    target_link_options(dirttown-vr-launcher PRIVATE -static -municode -s)
  endif()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.22)
  find_package(Threads REQUIRED)
  target_link_libraries(launcher-core PUBLIC ${CMAKE_DL_LIBS} Threads::Threads)
  add_executable(dirttown-vr-launcher main.cpp gui_gtk.cpp)
  target_link_libraries(dirttown-vr-launcher PRIVATE launcher-core PkgConfig::GTK3)
  target_link_options(dirttown-vr-launcher PRIVATE -static-libgcc -static-libstdc++ -s)
endif()
set_target_properties(dirttown-vr-launcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(MSVC)
  target_compile_options(launcher-core PRIVATE /utf-8)
  target_compile_options(dirttown-vr-launcher PRIVATE /utf-8)
else()
  target_compile_options(launcher-core PRIVATE -Os -ffunction-sections -fdata-sections)
  target_compile_options(dirttown-vr-launcher PRIVATE -Os -ffunction-sections -fdata-sections)
  target_link_options(dirttown-vr-launcher PRIVATE -Wl,--gc-sections)
endif()

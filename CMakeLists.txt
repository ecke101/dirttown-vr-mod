cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DTVR_VERSION LIMIT_COUNT 1)
project(dirttown_vr_mod VERSION ${DTVR_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only x64 game builds are supported.")
endif()
if(WIN32)
  add_compile_definitions(WINVER=0x0A00 _WIN32_WINNT=0x0A00)
endif()
if(LAUNCHER_DEPS)
  file(WRITE "${LAUNCHER_DEPS}/miniz-3.1.2/miniz_export.h" "#pragma once\n#define MINIZ_EXPORT\n")
  add_subdirectory(launcher)
endif()
if(NOT WIN32)
  return()
endif()
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
include_directories("${OPENXR_INCLUDE}" "${MINHOOK_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vr/temporal.hlsl" TEMPORAL_SOURCE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/temporal_shader.hpp" "#pragma once\nstatic const char temporalShader[]=R\"DT_TEMP(${TEMPORAL_SOURCE})DT_TEMP\";\n")
add_library(minhook STATIC "${MINHOOK_DIR}/src/buffer.c" "${MINHOOK_DIR}/src/hook.c"
  "${MINHOOK_DIR}/src/trampoline.c" "${MINHOOK_DIR}/src/hde/hde64.c")
add_library(dirttown-vr SHARED vr/inject.cpp)
target_include_directories(dirttown-vr PRIVATE "${LAUNCHER_DEPS}")
set_target_properties(dirttown-vr PROPERTIES PREFIX "")
target_link_libraries(dirttown-vr PRIVATE minhook d3d12 d3d11 dxgi d3dcompiler dxguid gdi32)
add_executable(vr-launch vr/launch.cpp)
add_executable(vr-probe vr/probe.cpp)
add_executable(temporal-test vr/temporal_test.cpp)
add_executable(core-test tests/core_test.cpp)
add_executable(eye-transfer-test tests/eye_transfer_test.cpp)
foreach(target vr-probe temporal-test core-test eye-transfer-test)
  target_link_libraries(${target} PRIVATE d3d12 d3d11 dxgi d3dcompiler dxguid gdi32)
endforeach()
target_include_directories(core-test PRIVATE vr)
target_include_directories(eye-transfer-test PRIVATE vr)
if(MINGW)
  foreach(target dirttown-vr vr-launch vr-probe temporal-test core-test eye-transfer-test)
    target_link_options(${target} PRIVATE -static -s)
  endforeach()
  target_link_options(vr-launch PRIVATE -municode)
endif()
if(MSVC)
  # The supported game build uses AVX; no hook relocates AVX instructions.
  target_compile_options(dirttown-vr PRIVATE /arch:AVX)
endif()
